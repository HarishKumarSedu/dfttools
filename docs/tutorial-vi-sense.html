VI Sense 1.5V Reference Measurement

A procedure-based test that powers up the device, configures the VI-sense analog test path, measures the internal 1.5V reference at a test pin, and validates it within ±5mV tolerance.

Test Objective

This test verifies the VI-sense block's 1.5V internal reference voltage. The reference is routed through the analog test path to the ADDR test pin, measured at IODATA1 with a multimeter.

💡
Stage: Tested

This test has been run on real silicon and results reviewed. The test uses four procedure calls: startup, global_enable, playback, and vi_sns_turn_on — demonstrating the reusable procedure pattern.

Complete Test Code

from dfttools import *
from time import sleep

# Import shared procedures
from Procedures.Startup        import startup
from Procedures.Global_enable  import global_enable
from Procedures.Playback       import playback
from Procedures.VI_SNS_turn_on import vi_sns_turn_on
from Procedures.VI_SNS_turn_offimport vi_sns_turn_off

def vis_1v5_meas():
    Test_Name = 'VIS_1V5_Meas'
    print(f'............ {Test_Name.lower()} ........')

    # ─ Step 1: Power-up sequence via shared procedures ─────────────
    startup()        # Force VDD=1.8V, PVDD=3.7V, init I2C
    global_enable() # Enable global chip features
    playback()      # Start audio playback (required for VI-sense)
    vi_sns_turn_on()# Enable VI-sense block

    # ─ Step 2: Settling time ────────────────────────────────────────
    sleep(0.0005)   # 500µs settling after VI-sense enable

    # ─ Step 3: Configure analog test path ──────────────────────────
    # Switch to page 1 for analog test registers
    I2C_WRITE(device_address=DEVICEADDR, field_info=i2c_page_sel_1, write_value=1)
    # Enable VI-sense analog test mux
    I2C_WRITE(device_address=DEVICEADDR, field_info=vis_atp_en, write_value=1)
    # Enable positive ATP connection (routes ref to ADDR pin)
    I2C_WRITE(device_address=DEVICEADDR, field_info=atp_p_en, write_value=1)
    # Enable negative ATP connection (routes GND ref to IODATA1)
    I2C_WRITE(device_address=DEVICEADDR, field_info=atp_n_en, write_value=1)
    # Select test signal: 1=REF_1V5 and Reference GND
    I2C_WRITE(device_address=DEVICEADDR, field_info=test_sel, write_value=1)

    # ─ Step 4: DC voltage measurement ──────────────────────────────
    expected_dc1 = 1.5      # Target: 1.5V reference
    dc1_error    = 0.005   # ±5mV tolerance

    # Measure IODATA1 with respect to ADDR (differential)
    vref1V5 = VMEASURE(signal=IODATA1, reference=ADDR,
                        expected_value=expected_dc1, error_spread=dc1_error)

    print(f'DC Reference Voltage: {vref1V5:.5f}V'
          f'  [Target: {expected_dc1:.2f}V ±{dc1_error*1000:.0f}mV]')

    # ─ Step 5: Pass/Fail evaluation ─────────────────────────────────
    if abs(vref1V5 - expected_dc1) <= dc1_error:
        print("PASS: Voltage within ±5mV specification")
    else:
        error_mv = abs(vref1V5 - expected_dc1) * 1000
        print(f"FAIL: Voltage error {error_mv:.1f}mV exceeds ±5mV limit")

    # ─ Step 6: Turn off measurement path, then VI-sense ────────────
    # Disconnect multimeter (force infinite resistance on measurement node)
    VMEASURE(signal=IODATA1, reference=GND, expected_value=float('Inf'))

    # Disable ATP connections
    I2C_WRITE(device_address=DEVICEADDR, field_info=atp_p_en, write_value=0)
    I2C_WRITE(device_address=DEVICEADDR, field_info=atp_n_en, write_value=0)
    I2C_WRITE(device_address=DEVICEADDR, field_info=vis_atp_en, write_value=0)
    I2C_WRITE(device_address=DEVICEADDR, field_info=test_sel, write_value=0)

    # Turn off VI-sense block
    vi_sns_turn_off()

if __name__ == "__main__":
    vis_1v5_meas()

Startup Procedure (Reused Across Tests)

# Procedures/Startup.py
from dfttools import *
from time import sleep

def startup():
    # Force all supply rails
    VFORCE(signal=VDD,    reference=GND, value=1.8)
    VFORCE(signal=RESETB, reference=GND, value=1.8)
    VFORCE(signal=PVDD,   reference=GND, value=3.7)
    sleep(0.2)  # Rail settling time

    # Initialize I2C registers
    I2C_WRITE(device_address=DEVICEADDR, field_info=i2c_page_sel, write_value=0x0)
    I2C_WRITE(device_address=DEVICEADDR, field_info=pll_mode, write_value=0x2)
    I2C_WRITE(device_address=DEVICEADDR, field_info=pll_ldet_cnt, write_value=0x10)

    # Switch to page 1 and unlock test registers
    I2C_REG_WRITE(device_address=DEVICEADDR, register_address=0xFE,
                  write_value=0x01, PageNo=1)
    I2C_REG_WRITE(device_address=DEVICEADDR, register_address=0x2F,
                  write_value=0xAA, PageNo=1)
    I2C_REG_WRITE(device_address=DEVICEADDR, register_address=0x2F,
                  write_value=0xBB, PageNo=1)

    # Return to page 0
    I2C_WRITE(device_address=DEVICEADDR, field_info=i2c_page_sel, write_value=0x0)
    I2C_WRITE(device_address=DEVICEADDR, field_info=tdm_fsyn_mnt_mode, write_value=0x0)
    I2C_WRITE(device_address=DEVICEADDR, field_info=iopad0_hyst, write_value=0x1)
    I2C_WRITE(device_address=DEVICEADDR, field_info=iodata1_hyst, write_value=0x1)

    # Read calibration measurement code
    cal_code = I2C_READ(device_address=DEVICEADDR, field_info=cal_meas,
                         expected_value=0x0)
    print(f"Calibration code: {hex(cal_code)}")

✅
Why VMEASURE(..., expected_value=float('Inf'))?

This is a DFTide convention for signaling the multimeter to open its measurement loop (disconnect its sense path) before reconfiguring the test MUX. In real hardware, the callback interprets infinity as a "disconnect" command.

// Quick Example

Bandgap trim in 15 lines

Sweep a 4-bit trim code over I²C, measure with VMEASURE, find optimal code — works in simulation with zero hardware.

VIEW FULL WALKTHROUGH →
BG_1V2_Trim.py
from dfttools import *
from time import sleep

target, lsb = 1.2, 7.57e-3
min_err, best = float('inf'), None

for i in list(range(8,16))+list(range(8)):
    I2C_WRITE(DEVICEADDR, otp_bg_trm, hex(i))
    v = VMEASURE("IODATA1","GND",
          expected_value=1.188+i*lsb,
          error_spread=0.025) - buf_offset
    err = abs(v-target)/target*100
    if err < min_err: min_err,best=err,hex(i)
    sleep(0.01)

I2C_WRITE(DEVICEADDR, otp_bg_trm, best)
print(f"Code={best} Err={min_err:.3f}%")

Boost Converter Test

A complete test for a DC-DC boost converter: power on with trimmed startup, verify output voltage accuracy at no-load and full-load, and check line regulation across the input voltage range.

Test Structure from JSON

The boost test is stored in Boost_1_.json. Multiple test functions cover different operating conditions:

Test NamePriorityDescription
Boost_Accuracy_No_LoadHighVerify output voltage at 0A load
Boost_Line_RegulationHighSweep input voltage, measure output stability
Boost_Load_RegulationHighSweep load current, measure output droop
Boost_EfficiencyMediumCalculate power conversion efficiency

Accuracy Test — No Load

from dfttools import *
from time import sleep
from Procedures.Startup_BUCK_force import startup_buck_force

def boost_accuracy_no_load():
    Test_Name = 'Boost_Accuracy_No_Load'
    print(f'............ {Test_Name} ........')

    # Nominal output voltage and limits
    typical_value = 5.0     # 5V nominal output
    high_limit    = 5.25   # +5% limit
    low_limit     = 4.75   # -5% limit
    error_spread  = 0.05   # Measurement noise for simulation

    # Start with BUCK force startup sequence (enables charge pump)
    startup_buck_force()
    sleep(0.05)  # 50ms settling at no-load

    # Measure boost output
    vboost = VMEASURE(signal="VBOOST", reference="GND",
                      expected_value=typical_value, error_spread=error_spread)

    print(f"  VBOOST = {vboost:.4f}V"
          f"  [Spec: {low_limit}–{high_limit}V]")
    error_pct = abs(vboost - typical_value) / typical_value * 100
    print(f"  Error  = {error_pct:.2f}%")

    if low_limit < vboost < high_limit:
        print(f'............ {Test_Name} PASSED ........')
    else:
        print(f'............ {Test_Name} FAILED ........')

    return vboost

Line Regulation Test

def boost_line_regulation():
    Test_Name = 'Boost_Line_Regulation'
    print(f'............ {Test_Name} ........')

    # Sweep input voltage from 3.0V to 4.2V (Li-Ion range)
    v_results = VFORCESWEEP(signal="VCC", reference="GND",
                             initial_value=3.0, end_value=4.2,
                             step=0.1, step_time=0.05)

    if not v_results:
        print("SKIP: No hardware callback registered for voltage sweep")
        return

    vout_min = min(v_results)
    vout_max = max(v_results)
    regulation = (vout_max - vout_min) / 5.0 * 100

    print(f"  VOUT range: {vout_min:.4f}V – {vout_max:.4f}V")
    print(f"  Line regulation: {regulation:.2f}%")

    if regulation < 2.0:
        print(f'............ {Test_Name} PASSED ........')
    else:
        print(f'............ {Test_Name} FAILED (regulation {regulation:.2f}%) ........')

Load Regulation Test

def boost_load_regulation():
    Test_Name = 'Boost_Load_Regulation'
    print(f'............ {Test_Name} ........')

    # Sweep load current from 0A to 1A
    load_results = AFORCESWEEP(signal="IOUT", reference="GND",
                                initial_value=0.0, end_value=1.0,
                                step=0.1, step_time=0.05)

    if not load_results:
        print("SKIP: No hardware callback")
        return

    vno_load = load_results[0]    # At 0A
    vfull_load = load_results[-1] # At 1A
    droop = vno_load - vfull_load
    regulation = droop / 5.0 * 100

    print(f"  V(no-load)   = {vno_load:.4f}V")
    print(f"  V(full-load) = {vfull_load:.4f}V")
    print(f"  Droop = {droop*1000:.1f}mV  ({regulation:.2f}%)")

    if regulation < 3.0:
        print(f'............ {Test_Name} PASSED ........')
    else:
        print(f'............ {Test_Name} FAILED ........')

Efficiency Calculation

def boost_efficiency():
    # Measure input power
    vin  = VMEASURE(signal="VCC",   reference="GND", expected_value=3.7)
    iin  = AMEASURE(signal="VCC",   reference="GND", expected_value=0.54)
    p_in = vin * iin

    # Measure output power
    vout = VMEASURE(signal="VBOOST", reference="GND", expected_value=5.0)
    iout = AMEASURE(signal="VBOOST", reference="GND", expected_value=0.39)
    p_out = vout * iout

    efficiency = p_out / p_in * 100
    print(f"Pin={p_in:.3f}W  Pout={p_out:.3f}W  η={efficiency:.1f}%")

    if efficiency > 85.0:
        print("PASS: Efficiency above 85%")
    else:
        print(f"FAIL: Efficiency {efficiency:.1f}% below 85% threshold")

Running All Boost Tests

if __name__ == '__main__':
    boost_accuracy_no_load()
    boost_line_regulation()
    boost_load_regulation()
    boost_efficiency()

Modular test structure

Each test function is independently callable. DFTide can run individual tests from the JSON database or chain them together into a full test suite. This modularity makes debugging and re-running specific tests fast and efficient.

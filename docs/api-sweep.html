API Reference

Force Sweep

Sweep force values (voltage, current, resistance, frequency) across a range with configurable step size and timing. Each step triggers a hardware measurement via the registered callback.

VFORCESWEEP — Voltage Sweep

VFORCESWEEP SWEEP voltage 
VFORCESWEEP(signal: str = 'VCC', reference: str = 'GND',
            initial_value: float = 0.0, end_value: float = 5.0,
            step: float = 0.1, step_time: float = 0.01) → list

Parameters

  • signal — Signal net to sweep
  • reference — Reference node
  • initial_value — Start value in volts
  • end_value — End value in volts
  • step — Step size in volts
  • step_time — Settling time between steps (seconds)

Returns

A list of measured values at each step, or an empty list if hardware callback is unavailable.

Example — Line Regulation Test

from dfttools import *

# Sweep VCC from 3.0V to 5.5V in 100mV steps, 10ms settling per step
results = VFORCESWEEP(signal="VCC", reference="GND",
                       initial_value=3.0, end_value=5.5,
                       step=0.1, step_time=0.01)

print(f"Measured {len(results)} data points")
for i, v in enumerate(results):
    print(f"  Step {i}: {v:.4f}V")

Example — Load Regulation Test

# Sweep load current from 0A to 2A — check output remains stable
load_results = AFORCESWEEP(signal="OUT+", reference="GND",
                            initial_value=0.0, end_value=2.0,
                            step=0.1, step_time=0.02)

vout_min = min(load_results)
vout_max = max(load_results)
regulation = (vout_max - vout_min) / vout_max * 100
print(f"Load regulation: {regulation:.2f}%")
print(f"VOUT range: {vout_min:.4f}V – {vout_max:.4f}V")

Sweep Callbacks

def voltage_force_sweep_callback(g, signal, reference, value, force_type):
    """
    Called at EVERY step during the sweep.
    Must return (hardware_available, measured_value).
    """
    # Set power supply to new voltage
    psu.write(f"VOLT {value}")
    # Wait for settling (the library also waits step_time)
    # Measure output at this voltage
    vout = float(dmm.query("MEAS:VOLT:DC?"))
    return True, vout

g.hardware_callbacks['voltage_force_sweep'] = voltage_force_sweep_callback

How Sweep Works Internally

def apply_force_and_measure_sweep(g, signal, reference,
                                    initial_value, end_value,
                                    step, step_time, force_type):
    sweep_results = []
    if g.hardware_callbacks[force_type]:
        current_value = initial_value
        while current_value <= end_value:
            hw_ok, measured = apply_force_and_measure(
                g, signal, reference, current_value, force_type)
            if hw_ok:
                sweep_results.append(measured)
                time.sleep(step_time)
            current_value += step
    return sweep_results

Bandgap Trim Sweep Example

A real-world use of sweep: iterating all trim codes to find the optimal bandgap reference setting:

from dfttools import *
from time import sleep

typical_value = 1.242  # Target bandgap reference
num_steps = 2**4        # 4-bit trim code (0..15)
step_size = 0.077625    # mV per trim step

min_error = float('inf')
optimal_code = None
optimal_value = None

for i in range(num_steps):
    # Write trim code to register
    I2C_WRITE(device_address=0x68, field_info=ref_vbg_trim,
              write_value=hex(i))
    sleep(0.1)

    # Measure resulting reference voltage
    expected = typical_value - 0.1241 + i * step_size
    measured = VMEASURE(signal="HWMute", reference="GND",
                         expected_value=expected, error_spread=0.025)
    error = abs(measured - typical_value) / typical_value * 100

    if error < min_error:
        min_error = error
        optimal_code = hex(i)
        optimal_value = measured

print(f"Optimal code:  {optimal_code}")
print(f"Optimal value: {optimal_value:.7f}V  (target={typical_value}V)")
print(f"Error:         {min_error:.4f}%")

Example 1 — Basic DC Accuracy Test

The simplest possible test: force a supply, measure an output, check limits.

from dfttools import *
from time import sleep

def a5v_accuracy_no_load():
    Test_Name = 'A5V_Accuracy_No_Load'
    print(f'............ {Test_Name} ........')

    typical_value = 4.8
    high_limit    = 4.9
    low_limit     = 4.7

    # Force input supply
    VFORCE(signal="VCC2+", reference="GND1-", value=14)
    sleep(0.05)

    # Measure A5V rail
    a5v = VMEASURE(signal="A5V", reference="GND1-",
                   expected_value=typical_value, error_spread=0.05)

    if low_limit < a5v < high_limit:
        print(f'............ {Test_Name} PASSED ........')
    else:
        print(f'............ {Test_Name} FAILED ........')

    error_pct = abs(a5v - typical_value) / typical_value * 100
    print(f"Measured: {a5v:.4f}V  Error: {error_pct:.3f}%")

if __name__ == '__main__':
    a5v_accuracy_no_load()

Example 2 — Bandgap Trim (Condensed)

from dfttools import *
from time import sleep

def trim_bg(target=1.242, num_bits=4, step_mv=0.077625):
    steps    = 2 ** num_bits
    best_err = float('inf')
    best_code, best_val = 0, None

    for i in range(steps):
        I2C_WRITE(device_address=0x68, field_info=ref_vbg_trim, write_value=hex(i))
        sleep(0.1)
        expected = target - (steps//2 - i) * step_mv
        measured = VMEASURE(signal="HWMute", reference="GND",
                             expected_value=expected, error_spread=0.025)
        err = abs(measured - target) / target * 100
        if err < best_err:
            best_err, best_code, best_val = err, hex(i), measured

    within_10pct = target*0.9 < best_val < target*1.1
    I2C_WRITE(device_address=0x68, field_info=ref_vbg_trim,
              write_value=best_code if within_10pct else 0)
    print(f"Trim: code={best_code}  val={best_val:.7f}V  err={best_err:.4f}%")
    print("PASS" if within_10pct else "FAIL")

Example 3 — Full Hardware Callback Setup

from dfttools import *
import pyvisa
from PyMCP2221A import PyMCP2221A

# ─ Open instruments ─────────────────────────────────────────────
rm  = pyvisa.ResourceManager()
psu = rm.open_resource('USB0::0x0957::0x0807::MY49012345::INSTR')
dmm = rm.open_resource('USB0::0x2A8D::0x1401::MY57229870::INSTR')

# ─ Force callbacks ──────────────────────────────────────────────
def v_force(g, signal, reference, value):
    psu.write(f"OUTP:VOLT {value}"); psu.write("OUTP ON")
    return True, float(psu.query("MEAS:VOLT?"))

# ─ Measure callbacks ────────────────────────────────────────────
def v_measure(signal, reference):
    try:
        return True, float(dmm.query("MEAS:VOLT:DC?"))
    except:
        return False, None

# ─ I2C callbacks ────────────────────────────────────────────────
def i2c_write(dev, reg, val, register):
    mcp = PyMCP2221A.PyMCP2221A(); mcp.I2C_Init()
    default = int(register['Default'], 16)
    mask    = int(register['Mask'],    16)
    lsb     = register['POS']
    byte    = ((default & ~mask) | (val << lsb)) & 0xFF
    mcp.I2C_Write(dev, bytearray([reg, byte]))
    return True

def i2c_read(dev, reg):
    mcp = PyMCP2221A.PyMCP2221A(); mcp.I2C_Init()
    mcp.I2C_Write(dev, bytearray([reg]))
    return mcp.I2C_Read(dev, 1)[0]

# ─ Register all callbacks ───────────────────────────────────────
g.hardware_callbacks = {
    'voltage_force':   v_force,
    'voltage_measure': v_measure,
    'i2c_write':       i2c_write,
    'i2c_read':        i2c_read,
}

Example 4 — Output Log Analysis

from dfttools import *
import json

# Run several operations
VFORCE(signal="VDD",   reference="GND", value=1.8)
VFORCE(signal="PVDD", reference="GND", value=3.7)
v1 = VMEASURE(signal="A5V",  reference="GND", expected_value=4.8)
v2 = VMEASURE(signal="VREF", reference="GND", expected_value=1.242)

# Print full execution log
print("── Execution Log ──")
print(json.dumps(g.output, indent=2))

# Filter to just measurements
measurements = [x for x in g.output if x['type'] == 'MEAS']
print(f"\nTotal measurements: {len(measurements)}")

Example 5 — Sweep with Analysis

from dfttools import *
import statistics

# Sweep PVDD and capture VOUT at each step
sweep_data = VFORCESWEEP(signal="PVDD", reference="GND",
                          initial_value=3.0, end_value=4.5,
                          step=0.1, step_time=0.02)

if sweep_data:
    print(f"Points measured: {len(sweep_data)}")
    print(f"Min:    {min(sweep_data):.4f}V")
    print(f"Max:    {max(sweep_data):.4f}V")
    print(f"Mean:   {statistics.mean(sweep_data):.4f}V")
    print(f"StdDev: {statistics.stdev(sweep_data)*1000:.2f}mV")
    regulation = (max(sweep_data) - min(sweep_data)) / 5.0 * 100
    print(f"Regulation: {regulation:.2f}%")
else:
    print("Simulation mode — no sweep data (no callback registered)")
